🔄 Bucles: for · for...in · for...of · while · forEach JS ▾
for clásico · while

  console.log(colores[i]);

  console.log(prop, empleado[prop]);
}

// for...of → VALORES del array
for (let color of colores) {
  console.log(color);
}

// forEach → array (índice opcional)
colores.forEach((c, i) => {
  console.log(i, c);
});

🧩 Métodos de Array map · filter · find · forEach · reduce · Set · spread
🔄 map( ) → transforma cada elemento JS ▾
map
// Crea NUEVO array aplicando función
const nums = [1, 2, 3];
const dobles = nums.map(n => n * 2);
// dobles → [2, 4, 6]

// Array de objetos → array de strings
const nombres = usuarios.map(u => u.nombre);

// Con índice
nums.map((n, i) => `${i}: ${n}`);
filter · find
// filter → NUEVO array con los que cumplen
const mayores = edades.filter(e => e >= 18);
// [15,22,18] → [22,18]

// find → PRIMER elemento que cumple
const usuario = usuarios.find(u => u.id === 2);
// → { id:2, nombre:"Luis" } o undefined
🔁 forEach · reduce · Set · Spread JS ▾
forEach · reduce
// forEach → solo ejecuta (no devuelve nada)
frutas.forEach(f => console.log(f));

// forEach con índice
frutas.forEach((f, i) => {
  console.log(i, f);
});

// reduce → acumula a UN solo valor
const total = notas.reduce((acum, n) => acum + n, 0);
// [7,8,9] → 24
Set · Spread (...)
// Set → elimina duplicados
const unicos = [...new Set([1,1,2,3,3])];
// unicos → [1, 2, 3]

// Spread → expande/clona
const juntos = [...a, ...b];
const copiaObj = { ...original, z: 3 };
const clonArr = [...original];

let tareas = [
  { id: 1, texto: "Estudiar", hecha: false }
];

function render() {
  const lista = document.getElementById("taskList");
  lista.innerHTML = "";
  tareas.forEach(t => {
    const li = document.createElement("li");
    li.textContent = t.texto;
    li.style.textDecoration = t.hecha
      ? "line-through" : "none";
    li.addEventListener("click", () => {
      t.hecha = !t.hecha;
      render(); // re-pintar
    });
    lista.appendChild(li);
  });
}

document.getElementById("addBtn")
  .addEventListener("click", () => {
  const input = document.getElementById("taskInput");
  tareas.push({
    id: Date.now(),
    texto: input.value,
    hecha: false
  });
  input.value = "";
  render();
});

render(); // inicial

let translations = {};
let lang = localStorage.getItem("lang") || "es";

fetch("translations.json")
  .then(r => r.json())
  .then(data => {
    translations = data;
    translatePage(lang);
  });

function translatePage(lang) {
  localStorage.setItem("lang", lang);
  document.querySelectorAll("[data-key]").forEach(el => {
    const key = el.dataset.key;
    if (translations[key]?.[lang]) {
      el.textContent = translations[key][lang];
    }
  });
}

document.getElementById("btn-es")
  .addEventListener("click", () => translatePage("es"));
document.getElementById("btn-en")
  .addEventListener("click", () => translatePage("en"));

🔐 Cifrado César — patrón típico de examen JS DOM ▾
HTML
<input type="text" id="frase">
<input type="range" id="number"
       min="0" max="20" value="10">
<span id="number-value">10</span>
<div id="solucion"></div>
<a id="run" href="#">Enviar</a>
JS
const alfabeto = "abcdefghijklmnñopqrstuvwxyzABCDEFGHIJKLMNÑOPQRSTUVWXYZ0123456789";

function cifrar(cadena, paso) {
  let resultado = "";
  for (let i = 0; i < cadena.length; i++) {
    const pos = alfabeto.indexOf(cadena[i]);
    if (pos === -1) {
      resultado += cadena[i]; // no está: igual
    } else {
      resultado += alfabeto[(pos + paso) % alfabeto.length];
    }
  }
  return resultado;
}

document.getElementById("run").addEventListener("click", () => {
  const paso = parseInt(document.getElementById("number").value);
  document.getElementById("solucion").textContent =
    cifrar(document.getElementById("frase").value, paso);
});

// Mostrar valor del range en vivo
document.getElementById("number")
  .addEventListener("input", () => {
  document.getElementById("number-value").textContent =
    document.getElementById("number").value;
});
